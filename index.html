    <script>

        const PRECIO_LADRILLO = 7531.30;

        const PRECIO_PUERTAS = 120000;

        const TOTAL_BASE = 110000;

        let totalDonado = TOTAL_BASE;

        let bricksData = [];

        let modoRayosX = false;

        function generarEstufa() {

            const contenedor = document.getElementById('estufa');

            contenedor.innerHTML = '';

            bricksData = [];

            for (let i = 0; i < 20; i++) {

                const fila = document.createElement('div');

                fila.className = 'fila';

                const cantidad = (i === 0 || i === 19) ? 6 : (i % 2 === 0 ? 5 : 6);

                for (let j = 0; j < cantidad; j++) {

                    const ladrillo = document.createElement('div');

                    ladrillo.className = 'ladrillo vacio';

                    ladrillo.dataset.index = bricksData.length;

                    bricksData.push({ funded: 0, donors: [] });

                    fila.appendChild(ladrillo);

                }

                contenedor.appendChild(fila);

            }

            bricksData[0].funded = PRECIO_LADRILLO;

            bricksData[0].donors = ['Moyi'];

            bricksData[1].funded = PRECIO_LADRILLO;

            bricksData[1].donors = ['Grupo'];

            bricksData[2].funded = PRECIO_LADRILLO * 0.5;

            bricksData[2].donors = ['Marian'];

            actualizarVisual();

        }

        function actualizarVisual() {

            const todos = document.querySelectorAll('.ladrillo');

            todos.forEach(el => {

                const idx = parseInt(el.dataset.index);

                const data = bricksData[idx];

                el.innerHTML = '';

                el.className = 'ladrillo';

                if (data.funded <= 0) {

                    el.className = 'ladrillo vacio';

                } else if (data.funded < PRECIO_LADRILLO) {

                    el.className = 'ladrillo parcial';

                    const porcentaje = (data.funded / PRECIO_LADRILLO) * 100;

                    el.style.setProperty('--progreso', `${porcentaje}%`);

                    el.textContent = data.donors.join('+');

                } else {

                    el.className = 'ladrillo';

                    el.textContent = data.donors.join('+');

                }

                if (modoRayosX) {

                    el.classList.add('interno');

                } else {

                    el.classList.remove('interno');

                }

            });

            const completos = bricksData.filter(b => b.funded >= PRECIO_LADRILLO).length;

            document.getElementById('ladrillosPuestos').textContent = completos;

            const financiado = bricksData.reduce((sum, b) => sum + b.funded, 0);

            totalDonado = TOTAL_BASE + financiado + (completos > 0 ? PRECIO_PUERTAS : 0);

            document.getElementById('totalRecaudado').textContent = '$' + totalDonado.toLocaleString('es-AR');

            const TOTAL_ESTIMADO = TOTAL_BASE + PRECIO_PUERTAS + bricksData.length * PRECIO_LADRILLO;

            const porcentaje = Math.min(100, Math.round((totalDonado / TOTAL_ESTIMADO) * 100));

            document.getElementById('relleno').style.width = porcentaje + '%';

            document.getElementById('relleno').textContent = porcentaje + '%';

        }

        function procesarDonacion() {

            const nombre = document.getElementById('nombre').value.trim();

            const monto = parseInt(document.getElementById('monto').value);

            if (!nombre) { mostrarToast('⚠️ Poné tu nombre'); return; }

            if (!monto || monto <= 0) { mostrarToast('⚠️ Ingresá un monto válido'); return; }

            cerrarModal();

            let restante = monto;

            let completados = 0;

            for (let i = 0; i < bricksData.length && restante > 0; i++) {

                const brick = bricksData[i];

                if (brick.funded >= PRECIO_LADRILLO) continue;

                const faltante = PRECIO_LADRILLO - brick.funded;

                const aporte = Math.min(restante, faltante);

                brick.funded += aporte;

                restante -= aporte;

                if (!brick.donors.includes(nombre)) brick.donors.push(nombre);

                if (brick.funded >= PRECIO_LADRILLO) completados++;

                if (brick.funded < PRECIO_LADRILLO) break;

            }

            let mensaje = `⚽ ¡${nombre} puso $${monto.toLocaleString()}!`;

            if (completados > 0) mensaje += ` ${completados} ladrillo(s) completado(s)`;

            else mensaje += ` Aportaste a un ladrillo parcial`;

            if (restante > 0) mensaje += ` (sobraron $${restante.toFixed(2)})`;

            mostrarToast(mensaje);

            actualizarVisual();

        }

        function verInterior() {

            modoRayosX = !modoRayosX;

            const estufa = document.getElementById('estufa');

            const imagen = document.getElementById('imagenRayosX');

            if (modoRayosX) {

                estufa.classList.add('girada');

                imagen.classList.add('visible');

                mostrarToast('🩻 Corte transversal activado');

            } else {

                estufa.classList.remove('girada');

                imagen.classList.remove('visible');

                mostrarToast('🧱 Vista normal');

            }

            const todos = document.querySelectorAll('.ladrillo');

            todos.forEach(l => {

                if (modoRayosX) {

                    l.style.opacity = '0.3';

                    l.style.background = 'rgba(255,100,0,0.4)';

                } else {

                    l.style.opacity = '1';

                    l.style.background = '';

                }

            });

            actualizarVisual();

        }

        function abrirModal() {

            document.getElementById('modalDonacion').style.display = 'flex';

            document.getElementById('nombre').focus();

        }

        function cerrarModal() {

            document.getElementById('modalDonacion').style.display = 'none';

            document.getElementById('nombre').value = '';

            document.getElementById('monto').value = '';

        }

        function mostrarToast(mensaje) {

            const toast = document.getElementById('toast');

            toast.textContent = mensaje;

            toast.style.opacity = '1';

            setTimeout(() => { toast.style.opacity = '0'; }, 2500);

        }

        generarEstufa();
